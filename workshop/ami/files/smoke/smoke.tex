\begin{theorem}\label{thm:main}
For every prime $p$ the ring $\mathbb{Z}_p$ is local with maximal ideal $\mathfrak{m} = p\mathbb{Z}_p$,
and $\mathscr{O}_X$ is a sheaf; see \cite{Serre}. Here $\llbracket a, b \rrbracket$ is an interval.
\end{theorem}
